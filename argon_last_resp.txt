1. Diagnoza techniczna (Dlaczego bot obraca się w trakcie kopania i marszu)

Przerwanie aktywnego kopania (Preemption Bug w BypassProcess): Skanery findNearestOre oraz findNearbyValuableDrop wykonywały się w każdym ticku nawet wtedy, gdy bot był w połowie rozbijania bloku (currentBreakingBlock != null). Wykrycie rudy lub dropu z boku natychmiast przerywało kopanie obecnego bloku, zmieniało state i szarpało kamerą w nową stronę.

Przełączanie celu w połowie rozbijania (BypassOreManager / BypassNavigator): Podczas ruchu o ułamek bloku algorytm w kolejnym ticku potrafił zmienić cel z frontHead na frontFeet lub z rudy na occluder przed zniszczeniem pierwszego bloku, co powodowało skakanie celownika.

Konflikt chodzenia z kopaniem i zaniżony zasięg: Brak rozdzielenia fazy czystego marszu (gdy trasa przed graczem jest wolna) od fazy bicia oraz zbyt niski próg zasięgu (2.8m–3.15m zamiast pełnego 3.8m) zmuszały bota do niepotrzebnego podchodzenia pod sam blok i kręcenia kamerą w trakcie biegu.

// W BypassProcess.java -> dodaj metodę sprawdzającą, czy bot jest w trakcie rozbijania bloku w reachu 3.8m:

public static final double HARD_MINING_REACH = 3.8D;
public static final double HARD_MINING_REACH_SQ = HARD_MINING_REACH * HARD_MINING_REACH; // 14.44

public boolean isCommittedToBreaking(LocalPlayer player, Level level) {
    if (currentBreakingBlock == null) return false;
    BlockState state = level.getBlockState(currentBreakingBlock);
    if (state.isAir() || !state.getFluidState().isEmpty() || state.getDestroySpeed(level, currentBreakingBlock) < 0.0F) {
        currentBreakingBlock = null;
        return false;
    }
    double distSq = player.getEyePosition().distanceToSqr(Vec3.atCenterOf(currentBreakingBlock));
    if (distSq > HARD_MINING_REACH_SQ) {
        currentBreakingBlock = null;
        return false;
    }
    return true;
}

BlockPos currentBreak = proc.getCurrentBreakingBlock();
if (currentBreak != null && !level.getBlockState(currentBreak).isAir()) {

        Optional<MiningAction> lockedAct = findMiningAction(proc, player, level, currentBreak, DIRECT_MINE_REACH);
        if (lockedAct.isPresent()) {
            proc.stopMovementInputs();
            return proc.mineBlockSafely(player, level, lockedAct.get().targetPos(), lockedAct.get().snapRotation());
        }
    }
}

// 2. ZASIĘG 3.8m: Jeśli ruda lub zasłaniający ją occluder mieści się w 3.8m, bij od razu z miejsca!
if (distSq3D <= DIRECT_MINE_REACH_SQ) {

// W BypassNavigator.java -> zastąp tickTunnelStep czystym podziałem WALK vs MINE (reach 3.8m):

public static PathingCommand tickTunnelStep(BypassProcess proc, LocalPlayer player, Level level, Direction tunnelDir) {
    BlockPos feetPos = player.blockPosition();
    BlockPos frontFeet = feetPos.relative(tunnelDir);
    BlockPos frontHead = frontFeet.above();

    // 1. Jeśli trasa bezpośrednio przed graczem (1-2 bloki) jest pusta -> 100% skupienia na marszu/sprincie!
    // Nie szukaj bloków do kopania z daleka i nie kręć kamerą, dopóki masz wolny korytarz przed sobą.
    BlockPos step2Feet = frontFeet.relative(tunnelDir);
    BlockPos step2Head = step2Feet.above();
    boolean immediatePathClear = level.getBlockState(frontHead).isAir() && level.getBlockState(frontFeet).isAir();
    boolean twoStepsClear = immediatePathClear && level.getBlockState(step2Head).isAir() && level.getBlockState(step2Feet).isAir();

    if (twoStepsClear) {
        return moveSafelyTowards(proc, player, level, Vec3.atBottomCenterOf(step2Feet), tunnelDir);
    }

    // 2. Wybór przeszkody blokującej trasę w zasięgu 3.8m (góra -> dół -> kolejny krok)
    BlockPos targetBlock = null;
    if (!level.getBlockState(frontHead).isAir()) {
        targetBlock = frontHead;
    } else if (!level.getBlockState(frontFeet).isAir()) {
        targetBlock = frontFeet;
    } else if (!level.getBlockState(step2Head).isAir()) {
        targetBlock = step2Head;
    } else if (!level.getBlockState(step2Feet).isAir()) {
        targetBlock = step2Feet;
    }

    // 3. Jeśli jest przeszkoda w reachu <= 3.8m -> skup się na jej rozbiciu do końca
    if (targetBlock != null) {
        Optional<MiningAction> act = BypassOreManager.findMiningAction(proc, player, level, targetBlock, 3.8D);
        if (act.isPresent()) {
            proc.stopMovementInputs();
            return proc.mineBlockSafely(player, level, act.get().targetPos(), act.get().snapRotation());
        }
    }

    // Fallback: podejdź do przodu po osi tunelu
    return moveSafelyTowards(proc, player, level, Vec3.atBottomCenterOf(frontFeet), tunnelDir);
}
